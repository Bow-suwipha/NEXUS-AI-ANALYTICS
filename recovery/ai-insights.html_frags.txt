from sqlalchemy import func
from datetime import datetime, timedelta

@app.get("/api/kpi")
def get_real_kpi(period: str = "month", db: Session = Depends(get_db)):
    # กำหนดช่วงเวลา (เช่น ดึงข้อมูลของเดือน 9 ปี 2026 ตาม Mock Data)
    # ในระบบจริงใช้ datetime.now() แต่เพื่อทดสอบกับข้อมูลจำลอง ขอตั้งเป็นวันที่ข้อมูลมีอยู่
    end_date = datetime(2026, 9, 30).date() 
    
    if period == "today":
        start_date = end_date
    elif period == "week":
        start_date = end_date - timedelta(days=7)
    else: # month
        start_date = end_date.replace(day=1)

    # 1. คำนวณยอดขายรวม และจำนวนออเดอร์
    sales_query = db.query(SalesData).filter(SalesData.sale_date.between(start_date, end_date))
    total_sales = db.query(func.sum(SalesData.total_price)).filter(SalesData.sale_date.between(start_date, end_date)).scalar() or 0
    total_orders = sales_query.count()

    # 2. หาสินค้าขายดีที่สุด
    best_product = db.query(
        SalesData.product_name, func.sum(SalesData.quantity).label('total_qty')
    ).filter(SalesData.sale_date.between(start_date, end_date)) \
     .group_by(SalesData.product_name).order_by(func.sum(SalesData.quantity).desc()).first()

    # 3. ข้อมูลสำหรับกราฟ (รายวัน)
    daily_sales = db.query(
        SalesData.sale_date, func.sum(SalesData.total_price)
    ).filter(SalesData.sale_date.between(start_date, end_date)) \
     .group_by(SalesData.sale_date).order_by(SalesData.sale_date).all()

    labels = [r[0].strftime("%d %b") for r in daily_sales]
    values = [float(r[1]) for r in daily_sales]

    return {
        "total_sales": float(total_sales),
        "total_orders": total_orders,
        "best_product": {
            "name": best_product[0] if best_product else "N/A",
            "quantity": int(best_product[1]) if best_product else 0
        },
        "chart": {"labels": labels, "values": values}
    }

        return {"reply": "กรุณาตั้งค่า Groq API Key ในหน้า Settings ก่อนใช้งานครับ"}

    # 2. ดึงข้อมูลสรุปจาก Database ไปเป็น Context ให้ AI รู้
    total_sales = db.query(func.sum(SalesData.total_price)).scalar() or 0
    context = f"ข้อมูลปัจจุบัน: ยอดขายรวมของบริษัทคือ {total_sales} บาท สินค้าหลักคือ Product A, B, C"

    try:
        client = Groq(api_key=setting.groq_api_key)
        chat_completion = client.chat.completions.create(
            messages=[
                {
                    "role": "system",
                    "content": f"คุณคือ NEXUS AI ผู้เชี่ยวชาญด้านยอดขาย ตอบสั้นๆ กระชับ เข้าใจง่าย นี่คือข้อมูลบริษัท: {context}"
                },
                {
                    "role": "user",
                    "content": request.message
                }
            ],
            model="llama3-70b-8192",
            temperature=0.5,
        )
        reply = chat_completion.choices[0].message.content
        return {"reply": reply}
    except Exception as e:
        return {"reply": f"เกิดข้อผิดพลาดในการเชื่อมต่อ Groq API: {str(e)}"}

import autogen
import pandas as pd

async def run_real_autogen_analysis(db: Session):
    # 1. เตรียม Data จาก DB
    sales_records = db.query(SalesData).all()
    df = pd.DataFrame([{
        "date": s.sale_date, "branch": s.branch, "product": s.product_name, "price": s.total_price
    } for s in sales_records])
    
    data_summary = df.groupby('branch')['price'].sum().to_string()

    # 2. ดึง API Key
    setting = db.query(SystemSetting).filter(SystemSetting.user_id == 1).first()
    if not setting or not setting.groq_api_key:
        await manager.broadcast("<span class='text-error'>Error: No Groq API Key found.</span>")
        return

    llm_config = {
        "config_list": [{"model": "llama3-70b-8192", "api_key": setting.groq_api_key, "base_url": "https://api.groq.com/openai/v1"}],
        "cache_seed": None
    }

    # 3. สร้าง Agents
    await manager.broadcast("<span class='text-emerald-400'>[System]</span> Initializing AutoGen Agents...")
    
    supervisor = autogen.UserProxyAgent(
        name="Supervisor",
        system_message="คุณคือผู้ควบคุม ตรวจสอบว่า Analyst ส่งข้อมูลมาครบหรือไม่",
        human_input_mode="NEVER",
        max_consecutive_auto_reply=1
    )
    
    analyst = autogen.AssistantAgent(
        name="DataAnalyst",
        system_message="คุณคือนักวิเคราะห์ข้อมูล วิเคราะห์ข้อมูลสาขาที่ได้และหาว่าสาขาไหนยอดตก พร้อมเสนอวิธีแก้สั้นๆ 1 ประโยค",
        llm_config=llm_config
    )

    await manager.broadcast(f"<span class='text-purple-400'>[Data Analyst]</span> Data loaded. Starting analysis on branch performance...")
    
    # 4. สั่งให้ AI คุยกัน
    chat_result = supervisor.initiate_chat(
        analyst,
        message=f"นี่คือข้อมูลยอดขายรวมแต่ละสาขา โปรดวิเคราะห์และสรุป Anomaly มาให้หน่อย:\n{data_summary}",
        summary_method="reflection_with_llm"
    )
    
    # ส่งข้อความผลลัพธ์ที่ AI คุยกันเสร็จแล้วกลับไปที่จอ
    ai_summary = chat_result.summary
    await manager.broadcast(f"<span class='text-yellow-400'>[Report Agent]</span> Analysis complete: {ai_summary[:100]}... (ดูฉบับเต็มในตาราง Insights)")
    
    # 5. บันทึกผลการวิเคราะห์ของ AI ลงตาราง AiInsight อัตโนมัติ
    new_insight = AiInsight(
        insight_type="Anomaly",
        title="AI Auto-Detected Insight",
        description=ai_summary,
        severity="High",
        related_branch="N/A"
    )
    db.add(new_insight)
    db.commit()
    
    await manager.broadcast("<span class='text-emerald-500 font-bold'>>> Task completed. Database updated.</span>")

@app.post("/api/run-ai")
async def trigger_real_ai(background_tasks: BackgroundTasks, db: Session = Depends(get_db)):
